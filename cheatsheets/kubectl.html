Kontext & Cluster

BefehlBeschreibung

Ressourcen anzeigen

BefehlBeschreibung
kubectl get podsPods im aktuellen Namespace
kubectl get pods -APods in allen Namespaces
kubectl get pods -o wideMit Knoten und IP
kubectl get pods -wÄnderungen live verfolgen
kubectl get allGängige Ressourcen des Namespace
kubectl get pods -l app=webNach Label filtern
kubectl describe pod nameDetails und Ereignisse eines Pods
kubectl get events --sort-by=.lastTimestampEreignisse chronologisch
kubectl get pod name -o yamlVollständige Definition als YAML
kubectl get pods -o jsonpath='{.items[*].metadata.name}'Gezielt Felder ausgeben
kubectl top pods / kubectl top nodesCPU-/RAM-Verbrauch (Metrics Server nötig)

Anwenden & Verwalten

BefehlBeschreibung
kubectl apply -f manifest.yamlRessourcen aus Datei erstellen/aktualisieren (deklarativ)
kubectl apply -f verzeichnis/Alle Manifeste eines Verzeichnisses
kubectl diff -f manifest.yamlUnterschied zum Cluster anzeigen
kubectl delete -f manifest.yamlRessourcen aus Datei entfernen
kubectl delete pod namePod löschen (Deployment startet ihn neu)
kubectl scale deployment web --replicas=3Anzahl der Repliken ändern
kubectl rollout status deployment webFortschritt eines Rollouts
kubectl rollout restart deployment webNeustart erzwingen (z. B. für neue Config)
kubectl rollout undo deployment webAuf vorherige Version zurückrollen
kubectl set image deployment/web app=image:2.0Image einer Anwendung aktualisieren
kubectl edit deployment webRessource direkt im Editor ändern
kubectl label pod name umgebung=testLabel setzen
kubectl annotate pod name hinweis="..."Annotation setzen

BefehlBeschreibung
kubectl logs podLogs eines Pods
kubectl logs pod -f --tail=100Live verfolgen, letzte 100 Zeilen
kubectl logs pod -c containerBestimmten Container wählen
kubectl logs pod --previousLogs des vorherigen (abgestürzten) Containers
kubectl exec -it pod -- shShell im Pod öffnen
kubectl exec pod -- envEinzelnen Befehl im Pod ausführen
kubectl port-forward pod 8080:80Lokalen Port auf einen Pod weiterleiten
kubectl port-forward svc/web 8080:80Auf einen Service weiterleiten
kubectl cp pod:/pfad/datei ./dateiDatei aus Pod kopieren
kubectl debug -it pod --image=busybox --target=containerDebug-Container an Pod anhängen
kubectl get pod name -o jsonpath='{.status.containerStatuses[*].state}'Container-Status (z. B. bei CrashLoopBackOff)
