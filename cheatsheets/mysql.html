Verbindung & Serverstatus

BefehlBeschreibung
mysql -u root -pAls root anmelden (Passwortabfrage)
mysql -u benutzer -p -h 127.0.0.1 -P 3306 datenbankMit Host, Port und Datenbank verbinden
mysql --defaults-file=~/.my.cnfZugangsdaten aus Datei (sicherer als Passwort in der Befehlszeile)
sudo mysqlAuf lokalen Servern oft per Socket-Auth ohne Passwort
mariadb -u benutzer -pClient bei MariaDB (Aufruf identisch)

Datenbanken & Tabellen

BefehlBeschreibung
CREATE DATABASE shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;Datenbank mit vollem Unicode anlegen
DROP DATABASE shop;Datenbank löschen
CREATE TABLE kunden (id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(255) UNIQUE, erstellt TIMESTAMP DEFAULT CURRENT_TIMESTAMP);Tabelle anlegen
DESCRIBE kunden; / SHOW CREATE TABLE kunden\GStruktur / vollständige Definition anzeigen
ALTER TABLE kunden ADD COLUMN telefon VARCHAR(30);Spalte hinzufügen
ALTER TABLE kunden MODIFY telefon VARCHAR(40);Spaltentyp ändern
ALTER TABLE kunden DROP COLUMN telefon;Spalte entfernen
CREATE INDEX idx_name ON kunden (name);Index anlegen
ALTER TABLE bestellungen ADD FOREIGN KEY (kunde_id) REFERENCES kunden(id);Fremdschlüssel
TRUNCATE TABLE kunden;Alle Zeilen schnell löschen (nicht rückgängig zu machen)
DROP TABLE kunden;Tabelle löschen

Abfragen & Daten

BefehlBeschreibung
SELECT * FROM kunden WHERE name LIKE 'M%' ORDER BY name LIMIT 10;Filtern, sortieren, begrenzen
SELECT land, COUNT(*) FROM kunden GROUP BY land HAVING COUNT(*) > 5;Gruppieren und Gruppen filtern
SELECT k.name, b.summe FROM kunden k JOIN bestellungen b ON b.kunde_id = k.id;Inner Join
SELECT k.name, b.summe FROM kunden k LEFT JOIN bestellungen b ON b.kunde_id = k.id;Left Join (auch Kunden ohne Bestellung)
INSERT INTO kunden (name, email) VALUES ('Max', 'max@example.org');Zeile einfügen
INSERT INTO kunden (name) VALUES ('A'), ('B'), ('C');Mehrere Zeilen
UPDATE kunden SET name = 'Maria' WHERE id = 1;Aktualisieren (immer mit WHERE!)
DELETE FROM kunden WHERE id = 1;Zeile löschen
START TRANSACTION; … COMMIT; / ROLLBACK;Transaktion abschließen / verwerfen
EXPLAIN SELECT * FROM kunden WHERE name = 'Max';Ausführungsplan (Indexnutzung prüfen)

Benutzer & Rechte

BefehlBeschreibung
CREATE USER 'app'@'localhost' IDENTIFIED BY 'Passwort';Benutzer anlegen ('%' = jeder Host)
GRANT SELECT, INSERT, UPDATE, DELETE ON shop.* TO 'app'@'localhost';Rechte auf eine Datenbank vergeben
GRANT ALL PRIVILEGES ON shop.* TO 'admin'@'localhost';Alle Rechte auf eine Datenbank
REVOKE INSERT ON shop.* FROM 'app'@'localhost';Recht entziehen
SHOW GRANTS FOR 'app'@'localhost';Rechte anzeigen
ALTER USER 'app'@'localhost' IDENTIFIED BY 'NeuesPasswort';Passwort ändern
FLUSH PRIVILEGES;Rechteänderungen sofort übernehmen
DROP USER 'app'@'localhost';Benutzer löschen

Backup & Wiederherstellung

BefehlBeschreibung
mysqldump -u root -p --single-transaction shop > shop.sqlEine Datenbank sichern (InnoDB ohne Sperren)
mysqldump -u root -p --all-databases > alle.sqlAlle Datenbanken sichern
mysqldump -u root -p --no-data shop > schema.sqlNur die Struktur sichern
mysqldump -u root -p shop | gzip > shop.sql.gzKomprimiert sichern
mysql -u root -p shop < shop.sqlAus Sicherung wiederherstellen
zcat shop.sql.gz | mysql -u root -p shopKomprimierte Sicherung einspielen
mariadb-dump / mariadb-importEntsprechende Werkzeuge bei MariaDB
SOURCE /pfad/shop.sql;SQL-Datei aus der MySQL-Konsole einlesen
mysqlcheck -u root -p --optimize --all-databasesTabellen prüfen und optimieren
