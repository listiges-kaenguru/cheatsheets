Schlüssel

Befehle für OpenSSL 3.x (openssl version). Die Option -noenc ersetzt das ältere -nodes (private Schlüssel unverschlüsselt speichern).

BefehlBeschreibung
openssl genpkey -algorithm ed25519 -out key.pemEd25519-Schlüssel erzeugen
openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out key.pemEC-Schlüssel (P-256)
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 -out key.pemRSA-Schlüssel mit 4096 Bit
openssl genpkey -algorithm RSA -aes-256-cbc -out key.pemSchlüssel mit Passwort schützen
openssl pkey -in key.pem -pubout -out pub.pemÖffentlichen Schlüssel extrahieren
openssl pkey -in key.pem -noout -textSchlüssel anzeigen
openssl pkey -in key.pem -out key-offen.pemPasswort vom Schlüssel entfernen
openssl pkey -pubout -in key.pem | openssl sha256Prüfwert des Schlüssels (mit Zertifikat vergleichen, siehe unten)

Zertifikate & CSRs

BefehlBeschreibung
openssl req -new -key key.pem -out anfrage.csr -subj "/CN=www.example.org"CSR zu vorhandenem Schlüssel erzeugen
openssl req -new -newkey ec -pkeyopt ec_paramgen_curve:P-256 -noenc -keyout key.pem -out anfrage.csr -subj "/CN=www.example.org" -addext "subjectAltName=DNS:www.example.org,DNS:example.org"Schlüssel und CSR mit SAN in einem Schritt
openssl req -x509 -newkey rsa:4096 -noenc -keyout key.pem -out cert.pem -days 365 -subj "/CN=server.local" -addext "subjectAltName=DNS:server.local"Selbstsigniertes Zertifikat
openssl req -in anfrage.csr -noout -text -verifyCSR anzeigen und Signatur prüfen
openssl x509 -in cert.pem -noout -textZertifikat vollständig anzeigen
openssl x509 -in cert.pem -noout -subject -issuer -datesInhaber, Aussteller, Gültigkeit
openssl x509 -in cert.pem -noout -ext subjectAltNameAlternative Namen (SAN)
openssl x509 -in cert.pem -noout -fingerprint -sha256Fingerabdruck
openssl x509 -in cert.pem -noout -checkend 2592000Läuft in den nächsten 30 Tagen ab? (Exit-Code 1 = ja)
openssl x509 -in cert.pem -noout -pubkey | openssl sha256Prüfwert des Zertifikats (muss zum Schlüssel passen)
openssl verify -CAfile ca-kette.pem cert.pemZertifikat gegen CA-Kette prüfen

Formate umwandeln

BefehlBeschreibung
openssl x509 -in cert.pem -outform der -out cert.derPEM → DER
openssl x509 -in cert.der -inform der -out cert.pemDER → PEM
openssl pkcs12 -export -out cert.pfx -inkey key.pem -in cert.pem -certfile kette.pemPEM → PKCS#12 (.pfx/.p12)
openssl pkcs12 -in cert.pfx -noenc -out alles.pemPKCS#12 → PEM (Schlüssel und Zertifikate)
openssl pkcs12 -in cert.pfx -nokeys -out cert.pemNur Zertifikate extrahieren
openssl pkcs12 -in alt.pfx -legacy -noenc -out alles.pemAlte PKCS#12-Dateien (RC2/3DES) öffnen
openssl crl2pkcs7 -nocrl -certfile kette.pem | openssl pkcs7 -print_certs -nooutAlle Zertifikate einer Kette auflisten

Verschlüsseln, Signieren & Hashen

BefehlBeschreibung
openssl enc -aes-256-cbc -pbkdf2 -iter 600000 -salt -in datei -out datei.encDatei mit Passwort verschlüsseln
openssl enc -d -aes-256-cbc -pbkdf2 -iter 600000 -in datei.enc -out dateiDatei entschlüsseln
openssl dgst -sha256 dateiSHA-256-Hash
openssl dgst -sha256 -sign key.pem -out datei.sig dateiDatei signieren
openssl dgst -sha256 -verify pub.pem -signature datei.sig dateiSignatur prüfen
openssl rand -base64 32Zufälliges Geheimnis (Base64)
openssl rand -hex 16Zufallswert (Hex)
openssl base64 -in datei / openssl base64 -d -in datei.b64Base64 kodieren / dekodieren
openssl passwd -6Passwort-Hash (SHA-512-crypt) für /etc/shadow erzeugen
